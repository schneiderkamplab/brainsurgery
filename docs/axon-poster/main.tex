\begin{minipage}[t]{0.49\linewidth}
\begin{tcolorbox}[posterbox,title={Why Axon?}]
\postertext{Modern checkpoints mix Python implementations, configuration
conventions, parameter naming schemes, and backend-specific execution code.
\axon{} separates the model-family description from those incidental details
while preserving tensor shapes and checkpoint paths as typed data.}
\vspace{3mm}
{\fontsize{21}{27}\selectfont
\begin{tabular}{@{}p{0.27\linewidth}p{0.65\linewidth}@{}}
\toprule
Challenge & Axon mechanism \\
\midrule
Model-family reuse & signatures, \code{do} blocks, pipes, scoped paths \\
Shape reasoning & \code{Tensor[B,S,D]}, \code{?Cache[B,H,T,DH]} \\
Checkpoint fidelity & parameter paths and config values remain first class \\
Backend portability & one typed graph contract for multiple runtimes \\
Inspection & rendered Axon remains readable at each source stage \\
\bottomrule
\end{tabular}
}
\end{tcolorbox}

\begin{tcolorbox}[posterbox,title={Full GPT-2 KV Axon Model}]
\begin{minipage}[t]{0.66\linewidth}
\vspace{0pt}
\begin{tcolorbox}[codebox]
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{18.4}{19.8}\selectfont]
{-# CHECKPOINTS ["gpt2", "openai-community/gpt2", "openai-community/gpt2-medium", "openai-community/gpt2-large", "openai-community/gpt2-xl"] #-}
{-# TASK "causal_lm" #-}

import Activations (gelu_new)
import Attention (attention, merge_heads, reshape_heads)
import Masking (causal_mask)
import Positions (position_ids)
import Cache (Cache, CacheLayer)
import Config
import NN
import Tensor

CONTEXT_SIZE = Config.dim @@n_positions default=1024
MODEL_DIM = Config.dim @@n_embd default=768
NUM_LAYERS = Config.dim @@n_layer default=12
NUM_HEADS = Config.dim @@n_head default=12

lin@path :: Path -> Tensor[B,S,D] -> Dim -> Tensor[B,S,O]
lin@path x dim = NN.linear@path x dim=dim bias=true transpose=true

gpt2_block :: Tensor[B,S,D] -> ?Tensor[B,K] -> ?CacheLayer[B,H,P,DH] -> (Tensor[B,S,D], ?CacheLayer[B,H,K,DH])
gpt2_block x attn_mask past_kv = do
  x1 <- NN.layernorm@ln_1 x
  a, new_kv <- scope@attn do
    q_lin, k_lin, v_lin <- lin@c_attn x1 dim=(3 * MODEL_DIM) |> Tensor.chunk parts=3
    q <- reshape_heads q_lin heads=NUM_HEADS
    k <- reshape_heads k_lin heads=NUM_HEADS
    v <- reshape_heads v_lin heads=NUM_HEADS
    k, v, new_kv <- Cache.update past_kv k v
    mask <- causal_mask q k window=CONTEXT_SIZE padding_mask=attn_mask
    a <- attention q k v mask |> merge_heads |> lin@c_proj dim=MODEL_DIM
    return a, new_kv
  x <- x + a
  xn <- NN.layernorm@ln_2 x
  m <- scope@mlp do
    return lin@c_fc xn dim=(4 * MODEL_DIM) |> gelu_new |> lin@c_proj dim=MODEL_DIM
  return x + m, new_kv

gpt2 :: Tensor.TokenIds[B,S] -> ?Tensor[B,K] -> ?Cache[B,H,P,DH] -> ?Bool -> (Tensor[B,S,V], ?Cache[B,H,K,DH])
gpt2 input_ids attn_mask past_kv use_cache = do
  tok <- NN.embedding@wte input_ids dim=MODEL_DIM
  pos <- position_ids input_ids attn_mask=attn_mask past_length=(Cache.past_length past_kv) pad_fill=1 |> NN.embedding@wpe
  x <- tok + pos
  new_kv <- use_cache ? Cache.init : null
  x, new_kv <- for@h i <- [0..NUM_LAYERS) carry (x, new_kv) do
    past_i <- Cache.index past_kv i
    x, new_i <- gpt2_block x attn_mask past_i
    new_kv <- Cache.append new_kv new_i
    yield x, new_kv
  logits <- NN.layernorm@ln_f x |> NN.linear@wte
  return logits, new_kv
\end{lstlisting}
\end{tcolorbox}
\end{minipage}

\vspace{0pt}
\begin{tcolorbox}[accentbox]
{\raggedright\fontsize{22}{28}\selectfont
\textbf{What this example shows.} This is a complete cached GPT-2 family model:
token and position embeddings feed a stack of attention/MLP blocks, the cache is
threaded through the loop, and the final projection reuses the input embedding
weight.\par\medskip

\textbf{Checkpoint paths stay in the source.} Calls such as \code{NN.embedding@wte},
\code{lin@c\_attn}, and \code{lin@c\_proj} name the tensors in the checkpoint.
The scopes \code{scope@attn}, \code{scope@mlp}, and \code{for@h} are resolved by
the compiler into absolute or templated parameter paths.\par\medskip

\textbf{Surface syntax is compact, not vague.} Pipes express data flow, multi-bind
statements unpack tensor tuples, optional cache arguments are explicit, and
\code{lin@path} is a small path-polymorphic wrapper around GPT-2 linear layers.\par
\medskip

\textbf{The lowered program is fully typed.} After elaboration and flattening,
every expression has a tensor/cache type, symbolic shape, arity, and path
metadata. The graph IR then carries this structured information to torch
codegen, the interpreter, pipeline execution, or future backends.\par}
\end{tcolorbox}
